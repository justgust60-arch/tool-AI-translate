import os
import asyncio
import whisper
import edge_tts
from deep_translator import GoogleTranslator

# ១. ទាញយកអត្ថបទចិន និងពេលវេលាចេញពីវីដេអូ (Speech-to-Text via Whisper)
def extract_chinese_subtitles(video_path):
    print("🎬 កំពុងវិភាគសំឡេងចិនក្នុងវីដេអូ...")
    model = whisper.load_model("base") # អាចដូរទៅ "small" ឬ "medium"
    result = model.transcribe(video_path, language="zh")
    return result["segments"]

# ២. បកប្រែអត្ថបទចិនទៅជាអក្សរខ្មែរ (Translation to Khmer)
def translate_zh_to_khmer(segments):
    print("🇰🇭 កំពុងបកប្រែជាភាសាខ្មែរ...")
    translator = GoogleTranslator(source='zh-CN', target='km')
    khmer_segments = []
    
    for seg in segments:
        zh_text = seg['text']
        km_text = translator.translate(zh_text)
        khmer_segments.append({
            'start': seg['start'],
            'end': seg['end'],
            'text': km_text
        })
        print(f"[{seg['start']:.1f}s - {seg['end']:.1f}s] {zh_text} -> {km_text}")
        
    return khmer_segments

# ៣. បង្កើតសំឡេងនិយាយខ្មែរ (Khmer Neural Edge-TTS)
async def generate_khmer_audio(text, output_mp3, voice="km-KH-SreymomNeural"):
    print(f"🎙️ កំពុងបង្កើតសំឡេង AI ខ្មែរ: {text[:20]}...")
    communicate = edge_tts.Communicate(text, voice)
    await communicate.save(output_mp3)

# ៤. បញ្ចូលសំឡេងខ្មែរ និង Subtitle ចូលវីដេអូ (FFmpeg Merger)
def merge_audio_video(video_path, audio_path, output_path):
    print("🎬 កំពុង Render វីដេអូចុងក្រោយ...")
    cmd = f'ffmpeg -i "{video_path}" -i "{audio_path}" -c:v copy -map 0:v:0 -map 1:a:0 "{output_path}" -y'
    os.system(cmd)
    print(f"✅ រក្សាទុកវីដេអូជោគជ័យ៖ {output_path}")
